if(UNIX AND NOT APPLE AND PJ_ENABLE_VECTOR_JCAN)
  find_package(PkgConfig)
  set(_pj_jcan_ok TRUE)
  set(_pj_jcan_skip_reason "")
  if(NOT PkgConfig_FOUND)
    set(_pj_jcan_ok FALSE)
    set(_pj_jcan_skip_reason "PkgConfig not found")
  else()
    pkg_check_modules(LIBUSB QUIET libusb-1.0)
    if(NOT LIBUSB_FOUND)
      set(_pj_jcan_ok FALSE)
      set(_pj_jcan_skip_reason "libusb-1.0 not found")
    endif()
  endif()
  # Prefer a version gate over check_cxx_source_compiles: project-wide -std=c++17
  # can make the <expected> probe fail even with g++-13 + -std=c++23.
  if(_pj_jcan_ok)
    if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS "12.0")
      set(_pj_jcan_ok FALSE)
      set(_pj_jcan_skip_reason "GCC ${CMAKE_CXX_COMPILER_VERSION} < 12 (need C++23 <expected>)")
    elseif(CMAKE_CXX_COMPILER_ID MATCHES "Clang" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS "16.0")
      set(_pj_jcan_ok FALSE)
      set(_pj_jcan_skip_reason "Clang ${CMAKE_CXX_COMPILER_VERSION} < 16 (need C++23 <expected>)")
    endif()
  endif()
  if(_pj_jcan_ok)
    message(STATUS "DataStreamVectorJcan enabled (compiler ${CMAKE_CXX_COMPILER_ID} ${CMAKE_CXX_COMPILER_VERSION}, libusb ok)")
    add_subdirectory(${CMAKE_SOURCE_DIR}/3rdparty/jcan_vector ${CMAKE_BINARY_DIR}/jcan_vector)
    add_subdirectory(DataStreamVectorJcan)
  else()
    message(WARNING "PJ_ENABLE_VECTOR_JCAN=ON but skipping DataStreamVectorJcan: ${_pj_jcan_skip_reason}")
  endif()
endif()
