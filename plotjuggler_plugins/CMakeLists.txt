if(UNIX AND NOT APPLE AND PJ_ENABLE_VECTOR_JCAN)
  find_package(PkgConfig)
  set(_pj_jcan_ok TRUE)
  if(NOT PkgConfig_FOUND)
    message(WARNING "PJ_ENABLE_VECTOR_JCAN=ON but PkgConfig not found; skipping DataStreamVectorJcan")
    set(_pj_jcan_ok FALSE)
  else()
    pkg_check_modules(LIBUSB QUIET libusb-1.0)
    if(NOT LIBUSB_FOUND)
      message(WARNING "PJ_ENABLE_VECTOR_JCAN=ON but libusb-1.0 not found; skipping DataStreamVectorJcan")
      set(_pj_jcan_ok FALSE)
    endif()
  endif()
  if(_pj_jcan_ok)
    include(CheckCXXSourceCompiles)
    # Prefer -std=c++23 flags so the probe works on CMake < 3.20 (no CXX23 dialect table).
    set(_pj_saved_required_flags "${CMAKE_REQUIRED_FLAGS}")
    set(CMAKE_REQUIRED_FLAGS "${CMAKE_REQUIRED_FLAGS} -std=c++23")
    check_cxx_source_compiles("#include <expected>\nint main() { std::expected<int,int> e{1}; return e.value(); }" PJ_HAS_CXX23_EXPECTED)
    set(CMAKE_REQUIRED_FLAGS "${_pj_saved_required_flags}")
    if(NOT PJ_HAS_CXX23_EXPECTED)
      message(WARNING "PJ_ENABLE_VECTOR_JCAN=ON but compiler lacks C++23 <expected>; skipping DataStreamVectorJcan")
      set(_pj_jcan_ok FALSE)
    endif()
  endif()
  if(_pj_jcan_ok)
    add_subdirectory(${CMAKE_SOURCE_DIR}/3rdparty/jcan_vector ${CMAKE_BINARY_DIR}/jcan_vector)
    add_subdirectory(DataStreamVectorJcan)
  endif()
endif()
